    \begin{definition}[Noetherian]
        A ring $R$ is called Noetherian if every ideal of $R$ is finitely generated.     
    \end{definition}

    \begin{proposition}
      Noetherian rings has equivalent statements
      
      \begin{itemize}
        \item Every ideal of $R$ is finitely generated, i.e. for an ideal $I$ of $R$, there exists $f_1, f_2, ..., f_n \in R$ such that
        \begin{align*}
          I = (f_1, f_2, \ldots, f_n)
        \end{align*}
        \item For every ascending chain
        \begin{align*}
          I_1 \subset I_2 \subset \ldots \subset I_m \subset \ldots  
        \end{align*}
        of ideal ring $R$ terminates, i.e. there exist $I_N$ such that $I_N = I_{N+1}$  
        \item Every non-empty set of ideal of $R$ has maximal element. 
      \end{itemize}
      
    \end{proposition}

    \begin{theorem}[Hilbert Basis Theorem]
        Let $R$ be a Noetherian ring. Then the polynomial ring $R[x]$ is also Noetherian.
    \end{theorem}

    \begin{theorem}[Weak Nullstellensatz, algebraic form]
      Suppose $k$ is an algebraically closed field, and let $\mathfrak{m}$ is maximal ideal of $k[x_1, \ldots, x_n]$, then we have
      \begin{align*}
        \mathfrak{m} = (x_1-a_1,\ldots, x_n-a_n)
      \end{align*}
    \end{theorem}

    \begin{proof}
      Let $\mathfrak{m}$ be maximal ideal of $k[x_1, \ldots, x_n]$, then we have $K = k[x_1, \ldots, x_n]/\mathfrak{m}$ is a field. Consider
      \begin{center}
        \begin{tikzcd}
          k \arrow[rd, "\varphi"] & \\
          k[x_1, \ldots, x_n] \arrow[r] & K 
        \end{tikzcd}
      \end{center}
      Notice that $\psi$ is field extension and since $k$ is algebraically closed, then $\psi$ is iso.   
      
    \end{proof}

    \begin{theorem}[Weak Nullstellensatz, geometric form]
      Suppose $k$ is an algebraically closed field. Let $I$ be a proper ideal in the polynomial ring $k[x_1, \ldots, x_n]$. Then we have 
      \begin{align*}
        V(I) \neq \emptyset
      \end{align*} 
      i.e., there exists a point $(a_1, \ldots, a_n) \in k^n$ such that $f(a_1, \ldots, a_n) = 0$ for all $f \in I$.
    \end{theorem}

    \begin{theorem}[Strong Nullstellensatz]
      Suppose $k$ is an algebraically closed field. Let $I$ be an ideal in the polynomial ring $k[x_1, \ldots, x_n]$. Then the ideal of all polynomials that vanish on the variety defined by $I$ is equal to the radical of $I$, i.e.,
      \begin{align*}
        I(V(I)) = \sqrt{I}.
      \end{align*}
    \end{theorem}
